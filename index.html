backend/server.js — Node.js + Express · runs on port 3001 
/**
 * ToolXr Backend Server
 * Handles: Word↔PDF, Excel→PDF, PPT→PDF, HTML→PDF, Bulk Convert, OCR
 * Requires: Node.js 18+, LibreOffice, Tesseract installed
 *
 * Install:  cd backend && npm install
 * Start:    node server.js
 * Dev:      npm run dev
 */

const express    = require('express');
const multer     = require('multer');
const cors       = require('cors');
const path       = require('path');
const fs         = require('fs');
const { exec }   = require('child_process');
const { promisify } = require('util');
const execAsync  = promisify(exec);

const app  = express();
const PORT = process.env.PORT || 3001;

// ── Middleware ────────────────────────────────────────────────────────────────
app.use(cors({ origin: '*' }));
app.use(express.json());

// Serve index.html as static frontend
app.use(express.static(path.join(__dirname, '..')));

// ── Upload config ─────────────────────────────────────────────────────────────
const UPLOAD_DIR = path.join(__dirname, 'uploads');
const OUTPUT_DIR = path.join(__dirname, 'outputs');
[UPLOAD_DIR, OUTPUT_DIR].forEach(d => fs.mkdirSync(d, { recursive: true }));

const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, UPLOAD_DIR),
  filename:    (req, file, cb) => {
    const unique = Date.now() + '-' + Math.round(Math.random() * 1e6);
    cb(null, unique + path.extname(file.originalname));
  }
});

const upload = multer({
  storage,
  limits: { fileSize: 50 * 1024 * 1024 }, // 50 MB max
  fileFilter: (req, file, cb) => {
    const allowed = [
      '.pdf','.docx','.doc','.xlsx','.xls','.pptx','.ppt',
      '.html','.htm','.md','.csv','.txt',
      '.jpg','.jpeg','.png','.webp','.gif','.bmp','.tiff','.avif'
    ];
    cb(null, allowed.includes(path.extname(file.originalname).toLowerCase()));
  }
});

// ── Helpers ───────────────────────────────────────────────────────────────────
async function libreofficeConvert(inputPath, outputDir, outputFormat = 'pdf') {
  const cmd = `libreoffice --headless --convert-to ${outputFormat} --outdir "${outputDir}" "${inputPath}"`;
  await execAsync(cmd, { timeout: 60000 });
  const base = path.basename(inputPath, path.extname(inputPath));
  const outFile = path.join(outputDir, `${base}.${outputFormat}`);
  if (!fs.existsSync(outFile)) throw new Error('Conversion produced no output');
  return outFile;
}

function cleanup(...files) {
  files.forEach(f => { try { if (f && fs.existsSync(f)) fs.unlinkSync(f); } catch (_) {} });
}

function sendAndClean(res, filePath, downloadName) {
  res.download(filePath, downloadName, err => {
    cleanup(filePath);
    if (err && !res.headersSent) res.status(500).json({ error: 'Download failed' });
  });
}

// ── Routes ────────────────────────────────────────────────────────────────────

// Health check
app.get('/api/health', (req, res) => {
  res.json({ status: 'ok', version: '1.0.0', tools: 38 });
});

// Word → PDF
app.post('/api/convert/word-to-pdf', upload.single('file'), async (req, res) => {
  if (!req.file) return res.status(400).json({ error: 'No file uploaded' });
  try {
    const out = await libreofficeConvert(req.file.path, OUTPUT_DIR);
    cleanup(req.file.path);
    sendAndClean(res, out, 'converted.pdf');
  } catch (e) { cleanup(req.file.path); res.status(500).json({ error: e.message }); }
});

// PDF → Word
app.post('/api/convert/pdf-to-word', upload.single('file'), async (req, res) => {
  if (!req.file) return res.status(400).json({ error: 'No file uploaded' });
  try {
    const out = await libreofficeConvert(req.file.path, OUTPUT_DIR, 'docx');
    cleanup(req.file.path);
    sendAndClean(res, out, 'converted.docx');
  } catch (e) { cleanup(req.file.path); res.status(500).json({ error: e.message }); }
});

// Excel → PDF

// HTML → PDF (wkhtmltopdf with LibreOffice fallback)
app.post('/api/convert/html-to-pdf', upload.single('file'), async (req, res) => {
  const htmlContent = req.body.html || (req.file ? fs.readFileSync(req.file.path, 'utf8') : null);
  if (!htmlContent) return res.status(400).json({ error: 'No HTML content' });
  const tmpHtml = path.join(UPLOAD_DIR, `tmp-${Date.now()}.html`);
  const outPdf  = path.join(OUTPUT_DIR, `page-${Date.now()}.pdf`);
  fs.writeFileSync(tmpHtml, htmlContent);
  try {
    await execAsync(`wkhtmltopdf "${tmpHtml}" "${outPdf}"`, { timeout: 30000 });
    cleanup(tmpHtml, req.file?.path);
    sendAndClean(res, outPdf, 'page.pdf');
  } catch {
    try {
      const out = await libreofficeConvert(tmpHtml, OUTPUT_DIR);
      cleanup(tmpHtml, req.file?.path);
      sendAndClean(res, out, 'page.pdf');
    } catch (e) {
      cleanup(tmpHtml, outPdf, req.file?.path);
      res.status(500).json({ error: 'Install wkhtmltopdf or LibreOffice: ' + e.message });
    }
  }
});

// Bulk Image Converter
app.post('/api/convert/img-convert', upload.array('files', 50), async (req, res) => {
  if (!req.files?.length) return res.status(400).json({ error: 'No files' });
  const targetFmt = req.body.format || 'pdf';
  const results = [];
  try {
    for (const file of req.files) {
      const out = await libreofficeConvert(file.path, OUTPUT_DIR, targetFmt);
      results.push(out);
      cleanup(file.path);
    }
    const archivePath = path.join(OUTPUT_DIR, `batch-${Date.now()}.zip`);
    await execAsync(`zip -j "${archivePath}" ${results.map(f => `"${f}"`).join(' ')}`);
    results.forEach(f => cleanup(f));
    sendAndClean(res, archivePath, 'converted-batch.zip');
  } catch (e) {
    req.files.forEach(f => cleanup(f.path));
    results.forEach(f => cleanup(f));
    res.status(500).json({ error: e.message });
  }
});

// OCR (Tesseract CLI)
app.post('/api/ocr', upload.single('file'), async (req, res) => {
  if (!req.file) return res.status(400).json({ error: 'No file' });
  const outBase = path.join(OUTPUT_DIR, `ocr-${Date.now()}`);
  try {
    await execAsync(`tesseract "${req.file.path}" "${outBase}" -l eng`, { timeout: 60000 });
    const text = fs.readFileSync(outBase + '.txt', 'utf8');
    cleanup(req.file.path, outBase + '.txt');
    res.json({ text });
  } catch (e) {
    cleanup(req.file.path);
    res.status(500).json({ error: 'Install Tesseract: sudo apt install tesseract-ocr — ' + e.message });
  }
});

// ── Auto-cleanup temp files older than 30 min ─────────────────────────────────
setInterval(() => {
  const cutoff = Date.now() - 30 * 60 * 1000;
  [UPLOAD_DIR, OUTPUT_DIR].forEach(dir => {
    fs.readdirSync(dir).forEach(file => {
      const p = path.join(dir, file);
      try { if (fs.statSync(p).mtimeMs < cutoff) fs.unlinkSync(p); } catch (_) {}
    });
  });
}, 10 * 60 * 1000);

// ── Start ──────────────────────────────────────────────────────────────────────
app.listen(PORT, () => {
  console.log(`\n🚀 ToolXr backend → http://localhost:${PORT}`);
  console.log(`📂 Uploads : ${UPLOAD_DIR}`);
  console.log(`📂 Outputs : ${OUTPUT_DIR}`);
  console.log(`\n🔧 Requires:`);
  console.log(`   • LibreOffice : sudo apt install libreoffice`);
  console.log(`   • Tesseract   : sudo apt install tesseract-ocr`);
  console.log(`   • wkhtmltopdf : sudo apt install wkhtmltopdf (optional)`);
  console.log(`\n🌐 Frontend  → http://localhost:${PORT}/index.html\n`);
});

/etc/nginx/sites-available/toolxr — production Nginx config 
# ToolXr — Nginx production config
# Replace yourdomain.com with your actual domain
#
# Setup steps:
#   sudo cp this-file /etc/nginx/sites-available/toolxr
#   sudo ln -s /etc/nginx/sites-available/toolxr /etc/nginx/sites-enabled/
#   sudo nginx -t && sudo systemctl reload nginx
#
# Then add HTTPS:
#   sudo certbot --nginx -d yourdomain.com -d www.yourdomain.com

server {
    listen 80;
    server_name yourdomain.com www.yourdomain.com;

    # Root points to your project folder
    root /var/www/toolxr;
    index index.html;

    # Serve the frontend
    location / {
        try_files $uri $uri/ /index.html;
        add_header X-Frame-Options "SAMEORIGIN";
        add_header X-Content-Type-Options "nosniff";
        add_header Referrer-Policy "no-referrer-when-downgrade";
    }

    # Forward /api/ requests to Node.js backend on port 3001
    location /api/ {
        proxy_pass         http://localhost:3001;
        proxy_http_version 1.1;
        proxy_set_header   Upgrade $http_upgrade;
        proxy_set_header   Connection 'upgrade';
        proxy_set_header   Host $host;
        proxy_set_header   X-Real-IP $remote_addr;
        proxy_cache_bypass $http_upgrade;
        client_max_body_size 50M;
        proxy_read_timeout   120s;
        proxy_connect_timeout 60s;
    }

    # Cache static assets
    location ~* \.(css|js|png|jpg|jpeg|gif|ico|svg|woff2)$ {
        expires 30d;
        add_header Cache-Control "public, no-transform";
    }

    # Block hidden files
    location ~ /\. {
        deny all;
    }

    # Gzip compression
    gzip on;
    gzip_types text/plain text/css application/json
               application/javascript text/xml application/xml
               application/xml+rss text/javascript;
    gzip_min_length 1000;
}

# Certbot will automatically add the HTTPS server block below
# after you run: certbot --nginx -d yourdomain.com
